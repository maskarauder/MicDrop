add_library(micdrop_media STATIC tone.cc)
target_include_directories(micdrop_media PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(micdrop_media PUBLIC cxx_std_20)
set_target_properties(micdrop_media PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(micdrop_media PRIVATE /W4 /WX)
else()
  target_compile_options(micdrop_media PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
