target_compile_options(micdrop_rtc PRIVATE /W4 /EHsc /bigobj)
set(MICDROP_RTC_LIBRARIES micdrop_windows_audio "${WEBRTC_ROOT}/lib/webrtc.lib"

target_link_libraries(micdrop_rtc PRIVATE ${MICDROP_RTC_LIBRARIES})
