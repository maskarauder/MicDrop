cmake_minimum_required(VERSION 3.22)

option(MICDROP_BUILD_TESTS "Build host-native contract tests" ON)
option(MICDROP_BUILD_RTC "Build the pinned native WebRTC shared library" OFF)

if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin" AND NOT CMAKE_SYSTEM_NAME STREQUAL "Android")
  if(NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.2" CACHE STRING "Minimum macOS deployment version")
  endif()
  if(NOT DEFINED CMAKE_OSX_SYSROOT)
    execute_process(
      COMMAND xcrun --sdk macosx --show-sdk-path
      OUTPUT_VARIABLE _micdrop_macos_sdk
      OUTPUT_STRIP_TRAILING_WHITESPACE
      ERROR_QUIET
    )
    if(_micdrop_macos_sdk AND EXISTS "${_micdrop_macos_sdk}")
      set(CMAKE_OSX_SYSROOT "${_micdrop_macos_sdk}" CACHE PATH "macOS SDK sysroot")
      if(NOT DEFINED CMAKE_SYSROOT)
        set(CMAKE_SYSROOT "${_micdrop_macos_sdk}")
      endif()
    endif()
  endif()
  if(NOT DEFINED CMAKE_OSX_ARCHITECTURES)
    set(CMAKE_OSX_ARCHITECTURES "arm64" CACHE STRING "macOS target architectures")
  endif()
endif()

if(MICDROP_BUILD_RTC AND NOT CMAKE_SYSTEM_NAME STREQUAL "Android")
  if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Linux")
    set(_micdrop_default_clang_dir "${CMAKE_CURRENT_SOURCE_DIR}/.deps/webrtc/linux/clang/bin")
  elseif(CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin")
    set(_micdrop_default_clang_dir "${CMAKE_CURRENT_SOURCE_DIR}/.deps/webrtc/macos/clang/bin")
  endif()
  if(_micdrop_default_clang_dir AND EXISTS "${_micdrop_default_clang_dir}/clang++")
    if(NOT DEFINED CMAKE_C_COMPILER)
      set(CMAKE_C_COMPILER "${_micdrop_default_clang_dir}/clang")
    endif()
    if(NOT DEFINED CMAKE_CXX_COMPILER)
      set(CMAKE_CXX_COMPILER "${_micdrop_default_clang_dir}/clang++")
    endif()
    if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin")
      if(NOT DEFINED CMAKE_OBJC_COMPILER)
        set(CMAKE_OBJC_COMPILER "${_micdrop_default_clang_dir}/clang")
      endif()
      if(NOT DEFINED CMAKE_C_COMPILER_TARGET)
        set(CMAKE_C_COMPILER_TARGET "aarch64-apple-darwin")
      endif()
      if(NOT DEFINED CMAKE_CXX_COMPILER_TARGET)
        set(CMAKE_CXX_COMPILER_TARGET "aarch64-apple-darwin")
      endif()
      if(NOT DEFINED CMAKE_OBJC_COMPILER_TARGET)
        set(CMAKE_OBJC_COMPILER_TARGET "aarch64-apple-darwin")
      endif()
    endif()
  endif()
endif()

project(MicDrop VERSION 0.1.0 LANGUAGES C CXX)

if(MICDROP_BUILD_TESTS AND NOT ANDROID)
  enable_testing()
endif()
if(MICDROP_BUILD_RTC AND MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
add_subdirectory(native/media)
if(NOT ANDROID)
  add_subdirectory(native/platform/windows)
endif()

if(MICDROP_BUILD_RTC)
  add_subdirectory(native/rtc)
endif()

if(MICDROP_BUILD_TESTS AND NOT ANDROID)
  add_subdirectory(tests/native)
endif()
